______________________________________________ ■ Table Facturas ______________________________________________

use [Auto Predio]
go

CREATE TABLE facturas (
    numero_factura INT IDENTITY(1000,1) PRIMARY KEY, -- empieza en 1000 y aumenta de 1 en 1
    fecha_emision DATETIME DEFAULT GETDATE(),
    forma_pago VARCHAR(30),
    total DECIMAL(15,2),
    estado VARCHAR(20), -- Ej: Pagada, Pendiente, Anulada
    id_cliente INT NOT NULL,
    id_vehiculo INT NULL, -- si aplica
    FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente),
    FOREIGN KEY (id_vehiculo) REFERENCES vehiculos(id_vehiculo)
);
